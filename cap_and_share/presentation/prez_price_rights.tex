\begin{frame}{How to achieve international climate cooperation?\label{proposal}}
\bbvsp
\ip Broad agreement on \blue{ability to pay} and \blue{responsibility} for past emissions: the \green{poorer or greener} a country, the \green{smaller the abatement burden} it should bear.
\ip Prominent economists and institutions have turned that precept into a concrete proposal --- \rose{differentiated carbon prices}:
  \bbvsp
  \ip Countries commit to a \blue{carbon price floor, increasing with their income level}, and each fully retains its own carbon revenue: \blue{no revenue is shared across borders}. \\
  \quad (Parry, Black \& Zhunussova, IMF, 2021; Wolfram et al., 2025; Banerjee, Duflo \& Greenstone, 2025)
  \ip \rose{Caveat}: in Banerjee et al., only low- and middle-income countries price; high-income ones instead fund damage compensation --- \rose{that transfer is the bargain}, and we do not model it. \\
  \ee
\ip However \rose{efficiency requires a uniform price}, and equity concerns can be settled by \rose{transfers}. % rather than through the price itself.
% \ip While this argument is standard --- or perhaps \textit{because} it is --- it has \rose{never been studied in detail}. That is the endeavour we tackle here.
\ee
\end{frame}

\bbvsp
% \ip In accordance with standard theory, we argue that 
\ip We study the rationales in favor of differentiated prices.
\ip We show that under standard assumptions, a \rose{global uniform price with transfers dominates differentiated prices}.

\section{Differentiated prices \textit{vs.} a uniform price}

\begin{frame}{Why differentiate? Four imperfections\label{literature}}
\small
\bbvsp
\ip \blue{Absent imperfections, a uniform price with transfers is optimal} (Aldy \& Stavins, 2012). %Under redistribution lower-income countries \rose{gain purchasing power}: allocated \blue{more rights than needs}, they could hold emissions constant and still receive a transfer --- while the high price rewards decarbonising with \rose{larger transfers still}.
LICs allocated more rights than emissions: they clearly gain. 
\ip Four imperfections do justify differentiation:
  \bbvsp
  \ip \rose{1. Divergent discount rates} (\green{Anthoff, Dennig \& Emmerling, 2021}), from misallocated capital. \\ But \blue{no model} yet \blue{tells} us \blue{whether} equalising prices through trading \blue{raises or lowers welfare}.
  \ip \rose{2. Country- or sector-specific distortive taxes} (\green{Babiker et al., 2004}; \green{Boeters, 2014}): if e.g. aviation stays under-taxed, a higher sectoral carbon price improves welfare. % Babiker Shows that in presence of existing distorsions (other taxes, e.g. on income or energy; terms of trade effects), a uniform price is not optimal: countries selling permits may lose compared to the situation where they have the same emissions rights but trading is not allowed. Optimality is restored if countries are allowed to trade permits but firms can only trade them domestically (thereby preventing a uniform price).
  \ip \rose{3. Market power in trade}: E.g. China, an oil importer, gains by taxing oil more --- depressing demand hence the world oil price. % --- and by pricing manufacturing emissions to raise the value of its exports.
  \ip \rose{4. Transfers politically infeasible}: Rationale asserted rather than established, by \green{Parry et al., 2021}; \green{Young-Brun et al., 2025}, \blue{Bauer et al.\ (2020)}'s \green{``efficiency--sovereignty'' trade-off}. \\ \pause
  $\Rightarrow$ \blue{Inconsistent with survey evidence} (\green{Fabre et al., 2025; Fabre, 2026}).
  \ee
\ip None of the four decisively discard a uniform price:  1. uncalibrated, 2. irrelevant, 3. compatible, 4. unproven. 
\ee
\end{frame}


\section{Standard Theory}

\begin{frame}{The static case\label{static_setting}}
\bbvs
\ip %Countries $i$, one period, no uncertainty. 
Country $i$ produces $y_i$, consumes $c_i$, and at carbon price $p_i$ faces abatement cost $a_i$:
\[ c_i = y_i - a_i + t_i, \qquad t_i \text{ = international transfer}. \]
\eed
\bbvs
\ip Welfare $u_i$ depends on $c_i$ and on \rose{global emissions} $E=\sum_j e_j n_j$, \rose{fixed at $E^*$} 
\bbvs \ip $n_i$ is population, $e_i$ emissions p.c. \ee
  % \bbvs \ip The question is \textit{not} ``what is the optimal target?'' but ``\rose{given the target, who pays, and how efficiently?}''
  % \ip Temperature is \textit{identical} under both regimes; only distribution and efficiency differ. \ee
\ip $p^*$: uniform price attaining $E^*$, i.e.\ \rose{$p^*$ s.t. $E^*=\sum_j e_j(p^*)n_j$}. \bbvs \ip Shorthands $e_i^*:=e_i(p^*)$, $a_i^*:=a_i(e_i^*)$. \ee
\ip A \blue{situation} for $i$ is described by $s=(t_i,p_i)$. Two anchors:
  \bbvs
  \ip \blue{Benchmark} $s^*=(0,p^*)$: uniform price, no transfer.
  \ip \blue{Autarky} $s^a=(0,p_i)$: differentiated price, no transfer.
  \ee
\ip $u_i$ can be expressed as function of $s$: we have $u_i(0,p_i)=c^a_i=y_i - a_i$, and $u_i(t,p^*) = c_i^* + t = y_i - a_i^* + t$.
\ee
\end{frame}


\begin{frame}{Step 1: the equivalent transfer\label{equivalent_variation}}
\small
\bbvs
\ip $V_i(p_i)$: the lump sum transfer which makes $i$ as well off with the global price as in autarky:
\[ u_i\bigl(V_i(p_i),\,p^*\bigr) = u_i(0,\,p_i) \qquad\Longrightarrow\qquad V_i(p_i) = c_i^a - c_i^* = a_i^* - a_i. \]
\eed
\bbvsp
\ip Optimising country sets $\frac{da_i}{de}(e_i^*)=-p^*$: Therefore, a first-order Taylor expansion around $e_i^*$ gives:
\[ \;V_i(p_i) \;\approx\; (e_i-e_i^*)\,p^* \]
\eed
\bbvsp
\ip Denoting $\varepsilon^*=-\tfrac{de_i/e_i^*}{dp/p^*} > 0$ the price elasticity at $p^*$, we have:
\[ \boxed{\;V_i(p_i)=\; (p^*-p_i)\,e_i^*\,\varepsilon^*\;}\]
\eed
\bbvsp
% \ip Sign is intuitive: if $p_i<p^*$, $i$ abates less in autarky $\Rightarrow$ moving it to $p^*$ requires \rose{compensation $V_i>0$}.
\ip Compensation needed is larger when \blue{price gap} is large, \blue{baseline emissions} are high, emissions are \blue{elastic}.
\ip \blue{Equivalent transfer $g_i^s$} of an arbitrary situation $s=(t_i,p_i)$, relative to the benchmark:
\[ g_i^s = \underbrace{t_i}_{\text{cash received}} + \underbrace{V_i(p_i)}_{\text{gain from facing } p_i \neq p^*} \]
\ee
% \bbvsp
% \ip One money-metric number at a common price $\Rightarrow$ comparable across regimes differing in \textit{both} $t$ and $p$. \hyperlink{equivalent_price}{\beamergotobutton{Why not an equivalent price?}}
% \ee
\end{frame}


\begin{frame}{Step 2: the correspondence between prices and rights\label{correspondence}}
\small
\bbvs
\ip Cap-and-trade at $p^*$, each person in $i$ endowed with rights $r_i$. Net monetary transfer to $i$:
\[ t_i^* = (r_i - e_i^*)\,p^* %\qquad (\text{$i$ \textit{sells} if } r_i>e_i^*, \text{ \textit{buys} otherwise}) 
\qquad\Longrightarrow\qquad g_i^* = t_i^*. \]
\eed
\bbvsp
\ip Autarky, from Step 1: $\;g_i^a = V_i(p_i) \approx (p^*-p_i)\,e_i^*\,\varepsilon^*$. \quad Setting $g_i^*=g_i^a$:
\[ \rose{\boxed{(r_i-e_i^*)\,p^* \;\approx\; (p^*-p_i)\,e_i^*\,\varepsilon^*}} \]
\[ \Longleftrightarrow\qquad
   \blue{r_i \approx \Bigl(1+\bigl(1-\tfrac{p_i}{p^*}\bigr)\varepsilon^*\Bigr)e_i^*}
   \qquad\Longleftrightarrow\qquad
   p_i \approx \Bigl(1+\tfrac{e_i^*-r_i}{e_i^*\varepsilon^*}\Bigr)p^*. \]
\eed
\bbvsp
\ip \rose{The key formula for policy design}: rights $r_i$ under uniform price equivalent to  autarky price $p_i$
  \bbvs
  \ip Low autarky price (\blue{$p_i<p^*$}) $\Rightarrow$ more rights than emissions \blue{$r_i>e_i^*$} to compensate the extra abatement cost.
  % \ip High autarky price ($p_i>p^*$) $\Rightarrow$ \blue{fewer rights} than equilibrium emissions suffice.
  \ee
\ip $\Rightarrow$ Any differentiated-price arrangement maps onto a cap-and-trade allocation.
\ip If the two can always be made equivalent, \textit{is there any reason to prefer one?} \quad Yes.
\ee
\end{frame}


\begin{frame}{Step 3: the uniform price weakly dominates, for \textit{every} country\label{efficiency}}
\bbvs
\ip Give country $i$ rights equal to its \textit{autarky} emissions, $r_i = e_i$. Net transfer $\tilde t_i = (e_i-e_i^*)p^*$.
\ip \rose{Convexity} of abatement costs implies:
\[ a_i - a_i^* \;\geq\; (e_i-e_i^*)\tfrac{da_i^*}{de} = -(e_i-e_i^*)p^*
\qquad\Longrightarrow\qquad \tilde t_i \;\geq\; a_i^*-a_i = V_i(p_i) = g_i^a. \]
\eed
\bbvsp
\ip Hence, consumption increases \textit{in every country} when trading is allowed and prices equalize:
\[ \boxed{\;\tilde c_i = y_i + \tilde t_i - a_i^* \;\geq\; y_i - a_i = c_i^a\;}  \]
\eed
\bbvsp
% \ip \blue{Mechanism}: in autarky a lower price cuts abatement costs, but \textit{at a diminishing rate}. Selling the spare permits instead earns the \rose{full market price $p^*$} on every tonne --- always at least as good a deal. % TODO: put back?
% \ip Strict dominance whenever abatement costs are strictly convex and prices differ across countries.
\ip This is \blue{Diamond \& Mirrlees (1971)} production efficiency applied to climate: absent other distortions, \rose{direct transfers beat differentiated factor prices as a redistributive instrument}.
\ee
\end{frame}


\begin{frame}{The dynamic case: the equivalence breaks\label{dynamic}}
\bbvs
\ip Add time $t$; each variable becomes a trajectory. $\beta_t$ = discount factor, $R_i=\sum_t r_{it}$ = $i$'s carbon budget.
\bbvs \ip $G_i = \sum_t g_{it}\beta_t$ : total discounted equivalent gain \ee
\ip Welfare equivalence now requires equality of \textit{entire discounted streams}, $G_i^U = G_i^{\mathcal{A}}$:
\[ \sum_t \beta_t\,(r_{it}-e_{it}^*)\,p_t^* \;\approx\; \sum_t \beta_t\,(p_t^*-p_{it})\,e_{it}^*\,\varepsilon_t^*. \]
\eed
\bbvsp
\ip In the static case $p^*$ was a \textit{common factor} and cancelled. Here it does not: the timing of allocation matters.
  \bbvs
  \ip A \blue{generous} allocation \blue{in} a \blue{low-price} period is \blue{worth less} than the same allocation when prices are high.
  \ip $\Rightarrow$ \rose{No direct correspondence between the intertemporal carbon budget $R_i$ and welfare}. %Timing matters, and timing depends on the full trajectory $(p_t^*)_t$.
  \ee
\ip \blue{Special case 1} --- homothetic prices, fixed emission rights shares: \rose{$p_{it}=\pi_i p_t^*$} and \rose{$r_{it}=\rho_i r_t$}
\[ \mathcal{B}:=\sum_t r_t\,\beta_t\,p_t^* \qquad  \rho_i\,\mathcal{B} - \sum_t e_{it}^*p_t^*\beta_t \;\approx\; (1-\pi_i)\sum_t e_{it}^*\varepsilon_t^*p_t^*\beta_t,  \]
\eed
\bbvsp
\ip Analogous to static formula, but $\mathcal{B}$ still requires the full price path $\Rightarrow$ \rose{not tractable without a model}.
\ee
\end{frame}


\begin{frame}{Special case 2: Hotelling --- and why it is the right benchmark\label{hotelling}}
\bbvs
\ip Hotelling rule: Fixed aggregate carbon budget + perfect foresight + efficient capital markets \\ $\Rightarrow$ \blue{carbon price rises at the rate of return on capital}. Discounting at that same rate, \rose{$\beta_t p_t^*=p_0$} is constant.
\ip $\beta_t p_t^*$ then factors out of both sums, and the dynamic condition collapses to:
\[ \boxed{\;R_i - E_i^* \;\approx\; (1-\pi_i)\sum_t e_{it}^*\,\varepsilon_t^*\;} \qquad E_i^*=\sum_t e_{it}^*\]
\eed
\bbvsp
\ip \rose{Structurally identical to the static formula}: the intertemporal carbon budget $R_i$ plays the role of the single-period allocation, and the homothetic price factor $\pi_i$ the role of $p_i/p^*$.
\ip Timing of the allocation no longer matters: every period carries the same discounted price.
\ip In the numerical analysis, we place ourselves in the Hotelling case. % TODO! do we?
% \ip \rose{This is the \textit{best possible} setting for the equivalence}: it stacks the most favourable assumptions (perfect foresight, efficient capital markets, fixed global budget, homothetic autarky prices).
% \ip $\Rightarrow$ \rose{If the approximation fails here, it fails everywhere}. Hence we take it to a calibrated model: not ``does it hold in theory?'' but ``\rose{how far do simulated welfare effects deviate, and what do the deviations reveal?}''
\ee
\end{frame}


\section{Numerical analysis}

\begin{frame}{Taking the formula to a calibrated model\label{nice}}
\bbvsp
\ip The formula rests on approximations a calibrated model need not satisfy: homogeneous elasticities, homothetic price paths, %perfect capital markets, 
a first-order Taylor expansion. %\rose{Whether it predicts well is an empirical question}.
% \ip We need a model that (i) has \blue{non-linear MAC curves}, (ii) lets autarky and global prices follow \blue{distinct multi-decade trajectories}, (iii) resolves welfare \blue{below the national level}. Representative-agent IAMs fail (iii).
\ip We use the model \rose{NICE} (Nested Inequalities Climate Economy, \green{Young-Brun et al., PNAS, 2025}): \\ \blue{DICE disaggregated at} the year-\blue{country-decile} level.
\ip \blue{Calibration}:
  \bbvsp
  \ip Horizon \blue{2030--2100}. Global price path $p_t^*$ enforces a strict \rose{1.8\textdegree{}C budget}: net-negative from 2080 onwards.
  \ip Welfare measure: NPV of \rose{Equally-Distributed-Equivalent (EDE) consumption}; discount rate 3\%, Atkinson inequality aversion $\eta=1.5$.
  \ip Grid search over $(\rho_i,\pi_i)$ for eight major regions: \blue{USA, Russia, EU27, China, Turkey, India, Nigeria, DRC}.
  \ee
\ee
\end{frame}


\begin{frame}{The two scenarios\label{scenarios}}
\bbvsp
\ip Choose a country $i$. Two scenarios:
\ip \blue{Autarky with differentiated prices} ($\pi_i$): country $i$ sets $p_i=\pi_i\,p^*$, zero transfer. %; \rose{all international pooling and transfers disabled}.
  \bbvsp
  \ip \blue{Revenue recycled domestically under Negishi} weighting (transfers $\propto$ inverse marginal utility) \\ $\Rightarrow$ neutralises the domestic redistribution component, \blue{isolating the pure cross-country effect}.
  % \ip The Rest of the World price $p_{-i}$ is estimated to keep the global cap. The linear identity $p^*=\omega_ip_i+(1-\omega_i)p_{-i}$ is \rose{not} enough: MAC curves are non-linear and emission shares endogenous $\Rightarrow$ the cap would be breached.
  % \ip $\Rightarrow$ \blue{Newton--Raphson} root-finding on $p_{-i}$ (bisection fallback), iterated to \rose{$<0.1\%$} relative error on the cap. \hyperlink{viability}{\beamergotobutton{Detail}}
  \ee
\ip \blue{Uniform price with varying rights} ($\rho_i$): everyone faces $p_t^*$; country $i$ receives \rose{$\rho_i \, n_{it}$} of global emission rights at $t$. 
  \bbvsp
  \ip RoW receives the \textit{residual} cap, grandfathered on baseline emissions. %: makes $\sum_j r_j = E^*$ hold exactly, and is status-quo neutral for the regions that are not the object of the exercise.
  \ee
% \ip \blue{Extensions}: abatement rates $\mu_i>1$ allowed (\rose{CDR}), MAC extrapolated beyond 100\%; \blue{viability filter} flagging RoW fossil subsidies ($\omega_i\pi_i>1$) and negative RoW rights ($\rho_i>1/pop_i$). \hyperlink{viability}{\beamergotobutton{Detail}}
\ee
\end{frame}

\begin{frame}{The proposals\label{proposal_floors}}
\small
\bbvsp
\ip \blue{IMF; Working Group on Climate Coalitions propose} a coalition of the willing with % members price carbon \blue{nationally}, with \rose{no cross-country revenue sharing}, under
 \blue{price floors differentiated by income}: \\ ~\blue{\$75/tCO$_\text{2}$ high-income}, \green{\$50/t upper-middle}, \red{\$25/t lower-income} (Parry et al., 2021; Wolfram et al., 2025).
\ip \blue{Wolfram et al.\ (2025)} narrow it further --- \blue{only carbon-intensive manufacturing} (steel, aluminium, cement, fertilisers --- $>$20\% of global CO$_\text{2}$), with a \blue{CBAM at the coalition's border} (at \$75/t):
  \bbvsp
  \ip The deal: the \green{EU renounces its own CBAM} in exchange for \green{China pricing its manufacturing emissions}, raising the price of Chinese final goods (Ch\^ateau et al., 2024).
  \ee
\ip Effect on global CO$_2$ emissions: \green{$-$1.7 to $-$2\%} (1.0--1.5\% of global GHG).
\ip \green{Little chance of being implemented}: China is $\sim$70\% of the \green{industrial emissions the coalition would cover}, yet appears unwilling to commit to \$50/t. (Economy-wide, members are $\sim$48\% of global CO$_\text{2}$, China $\sim$55\% of that.)

\section{Conclusion}

\begin{frame}{What the result does \textit{not} establish\label{limitations}}
\bbvsp
\ip \blue{A first-best statement}. The dominance result assumes away every other distortion. Where the second-best arguments bite --- market power, pre-existing sectoral taxes, divergent discount rates (\blue{Anthoff et al., 2022}; \blue{Babiker et al., 2004}) --- differentiation can be optimal in its own right, and we are \rose{silent}. \hyperlink{literature}{\beamergotobutton{The three arguments}}
\ip \blue{Negishi recycling is imposed}, not estimated. Domestic revenue returns in inverse proportion to marginal utility, neutralising the within-country distributive effect \textit{by construction}. That is what isolates the pure efficiency channel --- but it means our numbers say \rose{nothing about domestic incidence}, which is itself the contested political choice.
\ip \blue{The Rest of the World is a single aggregate}, grandfathered on baseline emissions, absorbing all adjustment through one price $p_{-i}$. \rose{Coalition formation, leakage and terms-of-trade effects among third countries are invisible} to us.
\ip \blue{One deviation at a time}. Each heatmap asks ``what if $i$ \textit{alone} goes it alone?''. The politically relevant case --- \rose{several large emitters differentiating simultaneously} --- is outside the grid.
\ip \blue{The dynamic equivalence leans on Hotelling}. Imperfect foresight or frictional capital markets break the constancy of $\beta_t p_t^*$ and \rose{reopen the timing problem} the static formula cannot see. \hyperlink{hotelling}{\beamergotobutton{Why Hotelling}}
\ee
\end{frame}


\begin{frame}{Conclusion\label{conclusion}}
\bbvsp
\ip Any schedule of \rose{differentiated carbon prices} has a \rose{welfare-equivalent uniform price plus rights allocation} --- and that allocation is \rose{weakly better for every country at once} (Diamond--Mirrlees).
\ip The equivalence \blue{breaks in a dynamic setting} but is restored under \blue{Hotelling}. NICE simulations show the static formula \rose{predicts $\rho_1$ well}: per-capita emissions relative to the world average, up to GE feedbacks (China) and exhausted cheap abatement (EU27).
\ip $\Rightarrow$ \rose{The political challenge is not efficiency, it is agreeing on the allocation.}
\ip But the heatmaps show a \rose{striking asymmetry}: the \red{red} region below the indifference curve is barely visible, rarely a few percentage points of EDE consumption NPV, while the \blue{blue} region above it reaches the scale ceiling for Nigeria and the DRC.
  \bbvsp \ip Being \textit{slightly} below the curve costs little; a generous allocation gains a lot, especially for low emitters. \rose{There is room for negotiated compromise at small efficiency cost.} \ee
\ip \blue{Caveats}: a first-best result, one deviating country at a time, RoW aggregated and grandfathered. \hyperlink{limitations}{\beamergotobutton{Limitations}}
\ip \begin{center} \textbf{Thank you for your attention!} \\ \blue{adrien.fabre@cnrs.fr} \quad \quad \quad \blue{constance.gorge@cired.fr} \end{center}
\ee
\end{frame}

\begin{frame}{Viability filter and the RoW price \label{viability} \hyperlink{scenarios}{\beamergotobutton{Go back}}}
\bbvs
\ip Two boundary conditions evaluated in \textit{every} year of 2030--2100:
\ip \blue{1. Market-distorting subsidies} ($\omega_i\times\pi_i>1$): the target country claims more than 100\% of the global price burden. To compensate, the model is forced to push $p_{-i}<0$ --- economically, a \rose{fossil-fuel subsidy for the RoW}.
\ip \blue{2. Negative emission rights} ($\rho_i>1/pop_i$): granting $i$ its multiplier uses more than the whole global budget, forcing the RoW into \rose{negative rights}. Binds for large populations: countries above 10\% of world population (China, India) trigger it for any $\rho_i>10$.
\ip \rule{\linewidth}{0.4pt}
\ip \blue{RoW price identity}, the starting point of the root-finding:
\[ p^* = \omega_i p_i + (1-\omega_i)p_{-i}, \quad \omega_i = e_i/E^* \qquad\Longrightarrow\qquad p_{-i} = p^*\,(E^*-e_i\pi_i)/e_{-i}. \]
\eed
\bbvs
\ip This linear approximation assumes emission-weighted price averaging delivers the same global abatement --- false with non-linear MACs, hence the \blue{Newton--Raphson} correction to $<0.1\%$ relative error on the cap.
\ee
\end{frame}

\begin{frame}{The equivalent price, and why we do not use it \label{equivalent_price} \hyperlink{equivalent_variation}{\beamergotobutton{Go back}}}
\bbvs
\ip One could instead define an \blue{equivalent price} $p_i^*$ such that $(0,p_i^*)$ is welfare-equivalent to $(t_i,p_i)$:
\[ t_i + (p^*-p_i)e_i^*\varepsilon^* \approx (p^*-p_i^*)e_i^*\varepsilon^* \qquad\Longrightarrow\qquad p_i^* \approx p_i - \frac{t_i}{e_i^*\varepsilon^*}. \]
\eed
\bbvs
\ip \rose{Problem}: $p_i^*$ can be \textit{negative} when the transfer is large, which has no natural interpretation as a carbon price.
\ip The equivalent transfer $g_i^s$ does not share this limitation $\Rightarrow$ we use it as the primary welfare metric.
\ee
\end{frame}
